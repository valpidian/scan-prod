const SIDEBAR_STATE_KEY = 'sidebar_sections';

function getSavedState() {
  try { return JSON.parse(localStorage.getItem(SIDEBAR_STATE_KEY)) || {}; }
  catch { return {}; }
}

function saveState(state) {
  localStorage.setItem(SIDEBAR_STATE_KEY, JSON.stringify(state));
}

function toggleSection(id) {
  const body = document.getElementById(`section-${id}`);
  const chevron = document.getElementById(`chevron-${id}`);
  const isCollapsed = body.classList.contains('collapsed');

  body.classList.toggle('collapsed', !isCollapsed);
  chevron.classList.toggle('collapsed', !isCollapsed);

  const state = getSavedState();
  state[id] = !isCollapsed ? 'collapsed' : 'open';
  saveState(state);
}

// Restaureaza starea la incarcare
document.addEventListener('DOMContentLoaded', function() {
  const state = getSavedState();
  ['workflow', 'notifications'].forEach(id => {
    if (state[id] === 'collapsed') {
      const body = document.getElementById(`section-${id}`);
      const chevron = document.getElementById(`chevron-${id}`);
      if (body) body.classList.add('collapsed');
      if (chevron) chevron.classList.add('collapsed');
    }
  });
});

  const btn = document.getElementById('btn-analyze');
  btn.disabled = true;
  btn.innerHTML = '<span class="spinner-border spinner-border-sm me-1"></span>Se analizeaza...';

  const resultDiv = document.getElementById('analyze-result');
  resultDiv.innerHTML = '<div class="text-muted small"><span class="spinner-border spinner-border-sm me-1"></span>Se parseaza sitemap-ul...</div>';

  fetch(`/scraping/${CODE}/analyze-sitemap`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ sitemap_url: sitemapUrl, url_filter: urlFilter, replace })
  })
  .then(r => r.json())
  .then(data => {
    btn.disabled = false;
    btn.innerHTML = '<i class="fas fa-search me-1"></i>Analizeaza';
    if (data.error) {
      resultDiv.innerHTML = `<div class="alert alert-danger py-2 small mb-0"><i class="fas fa-exclamation-triangle me-1"></i>${data.error}</div>`;
      return;
    }
    resultDiv.innerHTML = `
      <div class="alert alert-success py-2 small mb-0">
        <i class="fas fa-check-circle me-1"></i>
        <strong>${data.total}</strong> URL-uri gasite,
        <strong>${data.added}</strong> adaugate noi,
        <strong>${data.existing}</strong> existente deja.
        Se reincarca pagina...
      </div>`;
    setTimeout(() => location.reload(), 1200);
  })
  .catch(e => {
    btn.disabled = false;
    btn.innerHTML = '<i class="fas fa-search me-1"></i>Analizeaza';
    resultDiv.innerHTML = `<div class="alert alert-danger py-2 small mb-0">Eroare retea: ${e}</div>`;
  });
}

// ── TABEL URL-URI ──────────────────────────────────────────────────────────
function loadUrls(page) {
  urlsCurrentPage = page;
  const status = document.getElementById('filter-status')?.value || '';
  const tbody = document.getElementById('urls-tbody');
  if (!tbody) return;

  tbody.innerHTML = '<tr><td colspan="5" class="text-center py-2"><span class="spinner-border spinner-border-sm"></span></td></tr>';

  fetch(`/scraping/${CODE}/urls?page=${page}&status=${status}`)
  .then(r => r.json())
  .then(data => {
    if (!data.items.length) {
      tbody.innerHTML = '<tr><td colspan="5" class="text-center text-muted py-3">Niciun URL.</td></tr>';
      document.getElementById('urls-pagination').innerHTML = '';
      return;
    }
    const badge = {
      pending: '<span class="badge bg-warning text-dark">pending</span>',
      scraped: '<span class="badge bg-success">scraped</span>',
      error:   '<span class="badge bg-danger">error</span>',
    };
    tbody.innerHTML = data.items.map((u, i) => `
      <tr>
        <td class="text-muted small">${(page-1)*50+i+1}</td>
        <td>
          <div class="d-flex align-items-center gap-1">
            <a href="${u.url}" target="_blank" class="small text-truncate" style="max-width:340px;" title="${u.url}">${u.url}</a>
            <button type="button" class="btn btn-outline-secondary btn-sm btn-pill flex-shrink-0 copy-btn"
                    style="padding:1px 6px;font-size:.7rem;"
                    data-url="${u.url}" onclick="copyUrl(this)" title="Copiaza URL">
              <i class="fas fa-copy"></i>
            </button>
          </div>
          ${u.error_msg ? `<small class="text-danger">${u.error_msg}</small>` : ''}
        </td>
        <td>${badge[u.status] || u.status}</td>
        <td class="small">
          ${u.product_title
            ? `<span class="text-success">${u.product_title.substring(0,45)}</span><br><code class="text-muted" style="font-size:.68rem;">${u.product_sku||''}</code>`
            : '<span class="text-muted">—</span>'}
        </td>
        <td class="small text-muted">${u.scraped_at ? u.scraped_at.substring(0,16).replace('T',' ') : '—'}</td>
      </tr>`).join('');

    const pag = document.getElementById('urls-pagination');
    const start = (page-1)*50+1, end = Math.min(page*50, data.total);
    let h = `<span>${start}–${end} din ${data.total}</span><div class="d-flex gap-1 flex-wrap">`;
    for (let p = 1; p <= data.pages; p++) {
      if (data.pages > 8 && Math.abs(p-page) > 2 && p !== 1 && p !== data.pages) {
        if (p === 2 || p === data.pages-1) h += '<span class="px-1">…</span>';
        continue;
      }
      h += `<button class="btn btn-sm ${p===page?'btn-primary':'btn-outline-secondary'} btn-pill px-2 py-0" onclick="loadUrls(${p})">${p}</button>`;
    }
    pag.innerHTML = h + '</div>';
  });
}

// ── COPY URL ──────────────────────────────────────────────────────────────
function copyUrl(btn) {
  const url = btn.dataset.url;
  navigator.clipboard.writeText(url).then(() => {
    const orig = btn.innerHTML;
    btn.innerHTML = '<i class="fas fa-check"></i>';
    btn.classList.replace('btn-outline-secondary', 'btn-success');
    setTimeout(() => { btn.innerHTML = orig; btn.classList.replace('btn-success', 'btn-outline-secondary'); }, 1500);
  });
}

// ── AUTO-DISCOVER ──────────────────────────────────────────────────────────
const FIELD_MAP = { title:'sel_title', pret:'sel_pret', sku:'sel_sku', brand:'sel_brand', descriere:'sel_descriere', categorie:'sel_categorie' };
const FIELD_LABEL = { title:'Titlu', pret:'Pret', sku:'SKU', brand:'Brand', descriere:'Descriere', categorie:'Categorie' };
const SRC_BADGE = { structured:'<span class="badge bg-success ms-1" style="font-size:.58rem;">JSON-LD</span>', css:'<span class="badge bg-info text-dark ms-1" style="font-size:.58rem;">CSS</span>' };
window._discovered = {};

function discoverSelectors() {
  let url = document.getElementById('discover-url').value.trim();
  if (!url) {
    const firstLink = document.querySelector('#urls-tbody a');
    if (firstLink) { url = firstLink.href; document.getElementById('discover-url').value = url; }
    else { alert('Introdu un URL de produs.'); return; }
  }
  const btn = document.getElementById('btn-discover');
  btn.disabled = true;
  btn.innerHTML = '<span class="spinner-border spinner-border-sm me-1"></span>Se analizeaza...';

  fetch(`/scraping/${CODE}/discover`, {
    method: 'POST', headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ url })
  })
  .then(r => r.json())
  .then(data => {
    btn.disabled = false;
    btn.innerHTML = '<i class="fas fa-magic me-1"></i>Descopera automat';
    const div = document.getElementById('discover-result');
    div.classList.remove('d-none');
    if (data.error) { div.innerHTML = `<div class="alert alert-danger py-2 small">${data.error}</div>`; return; }

    // Salveaza cei mai buni selectori
    window._discovered = {};
    for (const [field, inputId] of Object.entries(FIELD_MAP)) {
      const res = data[field];
      if (res?.candidates?.length) window._discovered[inputId] = res.candidates[0].selector;
      else if (res?.selector)      window._discovered[inputId] = res.selector;
    }

    // Tabel aliniat cu tabelul de selectori
    let rows = '';
    for (const [field, inputId] of Object.entries(FIELD_MAP)) {
      const res = data[field];
      const src = SRC_BADGE[res?.source] || '';
      const val = res?.value
        ? `<span class="text-success" style="font-size:.75rem;" title="${res.value}">${res.value.substring(0,55)}${res.value.length>55?'…':''}</span>`
        : '<span class="text-muted" style="font-size:.75rem;">—</span>';
      let btns = '<span class="text-muted" style="font-size:.72rem;">Negasit</span>';
      if (res?.candidates?.length) {
        btns = res.candidates.map(c => {
          const sel = c.selector.replace(/\\/g,'\\\\').replace(/`/g,'\\`');
          return `<button type="button" class="btn btn-sm btn-outline-primary btn-pill me-1 mb-1"
            style="font-size:.68rem;font-family:monospace;padding:1px 7px;"
            onclick="applySelector('${inputId}',\`${sel}\`,this)"
            title="${c.preview.substring(0,100).replace(/"/g,'&quot;')}">${c.selector}</button>`;
        }).join('');
      }
      rows += `<tr>
        <td class="small fw-semibold">${FIELD_LABEL[field]}${src}</td>
        <td>${val}</td>
        <td>${btns}</td>
      </tr>`;
    }

    div.innerHTML = `
      <div class="border rounded bg-light mb-0">
        <div class="d-flex align-items-center justify-content-between px-3 py-2 border-bottom">
          <span class="small fw-semibold">Selectori descoperiti</span>
          <button type="button" class="btn btn-success btn-sm btn-pill" onclick="applyAllSelectors(this)">
            <i class="fas fa-magic me-1"></i>Populeaza automat toate campurile
          </button>
        </div>
        <table class="table table-sm mb-0">
          <thead class="table-light"><tr>
            <th style="width:110px;">Camp</th>
            <th style="width:180px;">Valoare gasita</th>
            <th>Selectori candidati</th>
          </tr></thead>
          <tbody>${rows}</tbody>
        </table>
      </div>`;
  })
  .catch(e => {
    btn.disabled = false;
    btn.innerHTML = '<i class="fas fa-magic me-1"></i>Descopera automat';
    const div = document.getElementById('discover-result');
    div.classList.remove('d-none');
    div.innerHTML = `<div class="alert alert-danger py-2 small">Eroare: ${e}</div>`;
  });
}

function applyAllSelectors(btn) {
  let count = 0;
  for (const [inputId, selector] of Object.entries(window._discovered)) {
    const input = document.getElementById(inputId);
    if (input && selector) {
      input.value = selector;
      input.classList.add('border-success');
      setTimeout(() => input.classList.remove('border-success'), 2000);
      count++;
    }
  }
  const orig = btn.innerHTML;
  btn.innerHTML = `<i class="fas fa-check me-1"></i>${count} campuri populate`;
  btn.disabled = true;
  setTimeout(() => { btn.innerHTML = orig; btn.disabled = false; }, 2000);
}

function applySelector(inputId, selector, btn) {
  const input = document.getElementById(inputId);
  if (input) { input.value = selector; input.classList.add('border-success'); setTimeout(() => input.classList.remove('border-success'), 1500); }
  btn.closest('.d-flex')?.querySelectorAll('button').forEach(b => { b.classList.remove('btn-primary'); b.classList.add('btn-outline-primary'); });
  btn.classList.remove('btn-outline-primary'); btn.classList.add('btn-primary');
}

// ── SCRAPING JOB ───────────────────────────────────────────────────────────
function startJob() {
  fetch(`/scraping/${CODE}/start`, { method:'POST', headers:{'Content-Type':'application/json'} })
  .then(r => r.json())
  .then(data => {
    if (data.error) { alert(data.error); return; }
    document.getElementById('run-idle').classList.add('d-none');
    document.getElementById('run-progress').classList.remove('d-none');
    progressOffset = 0;
    pollInterval = setInterval(pollStatus, 1500);
  });
}

function pollStatus() {
  fetch(`/scraping/${CODE}/status?offset=${progressOffset}`)
  .then(r => r.json())
  .then(data => {
    progressOffset = data.offset || progressOffset;
    const log = document.getElementById('progress-log');
    data.progress.forEach(p => {
      const line = document.createElement('div');
      line.className = p.status === 'error' ? 'text-danger' : 'text-success';
      line.textContent = `[${p.current}/${p.total}] ${p.status==='error' ? '✗ '+(p.error||p.url) : '✓ '+(p.title||p.url)}`;
      log.appendChild(line); log.scrollTop = log.scrollHeight;
    });
    if (data.progress.length) {
      const last = data.progress[data.progress.length-1];
      const pct = last.total ? Math.round(last.current/last.total*100) : 0;
      document.getElementById('progress-bar').style.width = pct+'%';
      document.getElementById('progress-counter').textContent = `${last.current}/${last.total} (${pct}%)`;
    }
    if (data.status === 'done') {
      clearInterval(pollInterval);
      document.getElementById('run-progress').classList.add('d-none');
      document.getElementById('run-done').classList.remove('d-none');
      const s = data.stats;
      document.getElementById('done-msg').innerHTML =
        `<i class="fas fa-check-circle me-2"></i>Finalizat: <strong>${s.inserted}</strong> inserate, <strong>${s.updated}</strong> actualizate, <strong>${s.errors}</strong> erori din <strong>${s.total}</strong>.`;
    } else if (data.status === 'error') {
      clearInterval(pollInterval);
      document.getElementById('run-progress').classList.add('d-none');
      document.getElementById('run-error').classList.remove('d-none');
      document.getElementById('error-msg').textContent = 'Eroare: ' + (data.error || 'necunoscuta');
    }
  });
}
